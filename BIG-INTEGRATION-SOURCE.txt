import UnifiedWorkspaceEntry from './components/UnifiedWorkspaceEntry';
import AIActionCenter from "./components/ai/AIActionCenter";
import { useState } from "react";
import "./App.css";

const API = "http://127.0.0.1:8001";

function App() {
  const [tab, setTab] = useState("dashboard");
  const [message, setMessage] = useState("");
  const [chat, setChat] = useState([]);
  const [status, setStatus] = useState("Checking...");
  const [file, setFile] = useState(null);

  async function checkAPI() {
    try {
      const res = await fetch(`${API}/health`);
      if (res.ok) {
        setStatus("ONLINE");
      } else {
        setStatus("API ERROR");
      }
    } catch {
      setStatus("OFFLINE");
    }
  }

  async function sendAgentMessage() {
    if (!message.trim()) return;

    const userMessage = message.trim();

    setChat((old) => [
      ...old,
      { role: "user", text: userMessage }
    ]);

    setMessage("");

    try {
      const response = await fetch(`${API}/chat`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          message: userMessage
        })
      });

      const data = await response.json();

      setChat((old) => [
        ...old,
        {
          role: "assistant",
          text:
            data.response ||
            data.message ||
            data.answer ||
            JSON.stringify(data)
        }
      ]);
    } catch (error) {
      setChat((old) => [
        ...old,
        {
          role: "assistant",
          text:
            "Agent API endpoint is not connected yet. Backend is running, but the chat route needs to be mapped."
        }
      ]);
    }
  }

  async function analyzeFile() {
    if (!file) return;

    const form = new FormData();
    form.append("file", file);

    try {
      const response = await fetch(`${API}/analyze`, {
        method: "POST",
        body: form
      });

      const data = await response.json();

      setChat((old) => [
        ...old,
        {
          role: "assistant",
          text: JSON.stringify(data, null, 2)
        }
      ]);
    } catch {
      setChat((old) => [
        ...old,
        {
          role: "assistant",
          text:
            "Data Analyst API is running, but the analyze endpoint needs to be mapped to the current backend route."
        }
      ]);
    }
  }

  return (
    <div className="app">
      <aside className="sidebar">
        <h1>Automation Studio</h1>
        <p className="subtitle">AI + Data Workspace</p>

        <button
          className={tab === "dashboard" ? "active" : ""}
          onClick={() => setTab("dashboard")}
        >
          Dashboard
        </button>

        <button
          className={tab === "analyst" ? "active" : ""}
          onClick={() => setTab("analyst")}
        >
          Data Analyst
        </button>

        <button
          className={tab === "agent" ? "active" : ""}
          onClick={() => setTab("agent")}
        >
          Personal AI Agent
        </button>

        <button
          className={tab === "system" ? "active" : ""}
          onClick={() => setTab("system")}
        >
          System
        </button>
      </aside>

      <main className="main">
        <header className="topbar">
          <div>
            <h2>
              {tab === "dashboard" && "Command Center"}
              {tab === "analyst" && "AI Data Analyst"}
              {tab === "agent" && "Personal AI Agent"}
              {tab === "system" && "System Status"}
            </h2>

            <p>
              Pandas ï¿½ DuckDB ï¿½ SQL ï¿½ Ollama ï¿½ Automation
            </p>
          </div>

          <div className="status">
            <span
              className={
                status === "ONLINE" ? "dot online" : "dot"
              }
            />
            {status}
          </div>
        </header>

        {tab === "dashboard" && (
          <section className="grid">
            <div className="card hero">
              <h3>Automation Studio</h3>
              <p>
                One workspace for data analysis, AI reasoning,
                automation and personal workflows.
              </p>

              <button onClick={checkAPI}>
                Check Backend
              </button>
            </div>

            <div className="card">
              <h3>Data Analyst</h3>
              <p>
                Excel / CSV ? Pandas ? DuckDB / SQL ? insights
              </p>
              <button onClick={() => setTab("analyst")}>
                Open Analyst
              </button>
            </div>

            <div className="card">
              <h3>Personal Agent</h3>
              <p>
                Ollama-powered assistant for personal workflows.
              </p>
              <button onClick={() => setTab("agent")}>
                Open Agent
              </button>
            </div>

            <div className="card">
              <h3>AI Engine</h3>
              <p>
                Local Ollama models are used as the reasoning layer.
              </p>
              <button onClick={() => setTab("system")}>
                Check System
              </button>
            </div>
          </section>
        )}

        {tab === "analyst" && (
          <section className="workspace">
            <div className="card large">
              <h3>Data Analyst Workspace</h3>

              <p>
                Upload a CSV or Excel file and send it to the
                Python analysis engine.
              </p>

              <input
                type="file"
                accept=".csv,.xlsx,.xls"
                onChange={(e) => setFile(e.target.files[0])}
              />

              {file && (
                <p className="file">
                  Selected: {file.name}
                </p>
              )}

              <button onClick={analyzeFile}>
                Analyze Dataset
              </button>
            </div>

            <div className="card">
              <h3>Analysis Stack</h3>

              <ul>
                <li>Python</li>
                <li>Pandas</li>
                <li>DuckDB</li>
                <li>SQL</li>
                <li>Excel</li>
                <li>AI insights</li>
              </ul>
            </div>
          </section>
        )}

        {tab === "agent" && (
          <section className="agent">
            <div className="card chat">
              <h3>Personal AI Agent</h3>

              <div className="messages">
                {chat.length === 0 && (
                  <div className="empty">
                    Ask the agent to analyze data,
                    explain something, plan a task,
                    or help with an automation.
                  </div>
                )}

                {chat.map((item, index) => (
                  <div
                    key={index}
                    className={
                      item.role === "user"
                        ? "message user"
                        : "message assistant"
                    }
                  >
                    <strong>
                      {item.role === "user"
                        ? "You"
                        : "Agent"}
                    </strong>

                    <pre>{item.text}</pre>
                  </div>
                ))}
              </div>

              <div className="inputRow">
                <input
                  value={message}
                  onChange={(e) => setMessage(e.target.value)}
                  onKeyDown={(e) => {
                    if (e.key === "Enter") {
                      sendAgentMessage();
                    }
                  }}
                  placeholder="Ask your personal AI agent..."
                />

                <button onClick={sendAgentMessage}>
                  Send
                </button>
              </div>
            </div>
          </section>
        )}

        {tab === "system" && (
          <section className="grid">
            <div className="card">
              <h3>Python API</h3>
              <p>{status}</p>

              <button onClick={checkAPI}>
                Test API
              </button>
            </div>

            <div className="card">
              <h3>Ollama</h3>
              <p>
                Local AI reasoning engine
              </p>
            </div>

            <div className="card">
              <h3>Database</h3>
              <p>
                DuckDB + SQL analysis layer
              </p>
            </div>

            <div className="card">
              <h3>Frontend</h3>
              <p>
                React + Vite
              </p>
            </div>
          </section>
        )}
        <AIActionCenter />
</main>
    
{/* UNIFIED_WORKSPACE_MOUNT */}
<div id="unified-workspace-panel" style={{width:"100%", marginTop:"24px"}}>
  <UnifiedWorkspaceEntry />
</div>
{/* END_UNIFIED_WORKSPACE_MOUNT */}</div>
  );
}

export default App;

const API_BASE = "http://127.0.0.1:8001";
const OLLAMA_BASE = "http://127.0.0.1:11434";

export async function checkPythonAPI() {
  const response = await fetch(`${API_BASE}/health`);
  if (!response.ok) {
    throw new Error("Python API health check failed");
  }
  return response.json();
}

export async function checkOllama() {
  const response = await fetch(`${API_BASE}/health`);
  if (!response.ok) {
    throw new Error("Ollama health check failed");
  }
  return response.json();
}

export async function askOllama(prompt, model) {
  const response = await fetch(`${OLLAMA_BASE}/api/generate`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify({
      model,
      prompt,
      stream: false
    })
  });

  if (!response.ok) {
    throw new Error(`Ollama request failed: ${response.status}`);
  }

  return response.json();
}
export async function discoverAPI() {
  const response = await fetch(`${API_BASE}/health`);

  if (!response.ok) {
    throw new Error("Unable to read Python API contract");
  }

  return response.json();
}

export async function getAIStackStatus() {
  const result = {
    pythonApi: "offline",
    ollama: "offline"
  };

  try {
    const apiResponse = await fetch("http://127.0.0.1:8001/health");
    result.pythonApi = apiResponse.ok ? "online" : "error";
  } catch {
    result.pythonApi = "offline";
  }

  try {
    const ollamaResponse = await fetch("http://127.0.0.1:11434/api/tags");
    result.ollama = ollamaResponse.ok ? "online" : "error";
  } catch {
    result.ollama = "offline";
  }

  return result;
}
